// Generates the pimax-voice e2e fixture with Kokoro-82M (local TTS): English + Spanish utterances with noise between.
use pimax_voicebench::resample::resample;
use pimax_voicebench::sherpa::{default_lib_path, Lib, Tts, TtsConfig};
use pimax_voicebench::wav;
use std::path::PathBuf;
fn main() {
    let out = PathBuf::from(std::env::args().nth(1).expect("OUT.wav"));
    let home = PathBuf::from(std::env::var("HOME").unwrap());
    let kokoro = home.join("models/kokoro-multi-lang-v1_0");
    let lib = Lib::load(&default_lib_path()).unwrap();
    let en = Tts::new(&lib, &TtsConfig::from_dir(&kokoro, "lexicon-us-en.txt,lexicon-zh.txt", "", 4)).unwrap();
    let es = Tts::new(&lib, &TtsConfig::from_dir(&kokoro, "", "es", 4)).unwrap();
    let items: [(&Tts, i32, &str); 3] = [
        (&en, 3, "Recenter the view."),
        (&es, 29, "Hola, esto es una prueba del micrófono."),
        (&es, 28, "Borra la pantalla."),
    ];
    let mut s: Vec<f32> = vec![0.0; 12_800];
    let mut marks = Vec::new();
    for (tts, spk, text) in items {
        let a = tts.generate(text, spk, 1.0, &mut |_| true).unwrap();
        let a16 = resample(&a.samples, a.sample_rate, 16_000);
        marks.push((s.len(), s.len() + a16.len(), text));
        s.extend(a16);
        s.extend(vec![0.0; 19_200]);
    }
    // light noise everywhere (about -62 dBFS RMS)
    let mut x = 12345u32;
    for v in s.iter_mut() {
        x = x.wrapping_mul(1664525).wrapping_add(1013904223);
        *v += 0.0014 * ((x >> 8) as f32 / (1u32 << 24) as f32 * 2.0 - 1.0);
    }
    wav::write_pcm16(&out, &s, 16_000, 1).unwrap();
    for (a, b, t) in marks { println!("{:.3} {:.3} {t}", a as f64 / 16e3, b as f64 / 16e3); }
    println!("total {:.2} s", s.len() as f64 / 16e3);
}
